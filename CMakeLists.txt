cmake_minimum_required(VERSION 3.28)
project(obs-streambridge VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(STREAMBRIDGE_BUILD_PLUGIN "Build the Windows OBS plugin" ON)
option(STREAMBRIDGE_INTEGRATION_TESTS "Build libobs integration runner" OFF)
set(STREAMBRIDGE_GITHUB_REPOSITORY "AugustusNobilis/StreamBridge" CACHE STRING "Public GitHub owner/repository for opt-in updates")
if(STREAMBRIDGE_GITHUB_REPOSITORY AND NOT STREAMBRIDGE_GITHUB_REPOSITORY MATCHES "^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "STREAMBRIDGE_GITHUB_REPOSITORY must be owner/repository.")
endif()
add_library(streambridge-core STATIC src/core/retry.cpp src/core/redact.cpp)
target_include_directories(streambridge-core PUBLIC src)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
if(BUILD_TESTING)
  add_executable(core-tests tests/core-tests.cpp)
  target_link_libraries(core-tests PRIVATE streambridge-core)
  add_test(NAME core COMMAND core-tests)
endif()
if(NOT STREAMBRIDGE_BUILD_PLUGIN)
  return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The plugin requires Windows x64.")
endif()
add_library(streambridge-bootstrap STATIC updater/core.cpp)
target_include_directories(streambridge-bootstrap PUBLIC updater)
target_compile_definitions(streambridge-bootstrap PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(streambridge-bootstrap PUBLIC winhttp bcrypt shell32 ole32 advapi32 windowsapp)
add_executable(updater WIN32 updater/main.cpp updater/updater.rc)
target_link_libraries(updater PRIVATE streambridge-bootstrap comctl32)
set_target_properties(streambridge-bootstrap updater PROPERTIES
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(BUILD_TESTING)
  add_executable(bootstrap-tests tests/bootstrap-tests.cpp)
  target_link_libraries(bootstrap-tests PRIVATE streambridge-bootstrap)
  set_target_properties(bootstrap-tests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME bootstrap COMMAND bootstrap-tests)
endif()
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/sdk"
  "${CMAKE_SOURCE_DIR}/.deps/obs-deps-2025-08-23-x64"
  "${CMAKE_SOURCE_DIR}/.deps/obs-deps-qt6-2025-08-23-x64")
find_package(libobs 32 REQUIRED CONFIG)
find_package(obs-frontend-api REQUIRED CONFIG)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Network)
find_package(Qt6QSchannelBackendPlugin REQUIRED CONFIG PATHS "${Qt6Network_DIR}" NO_DEFAULT_PATH)
add_library(streambridge-accounts STATIC
  src/config/config.cpp src/security/secret-store.cpp
  src/oauth/http.cpp src/oauth/loopback.cpp src/oauth/account.cpp)
set_target_properties(streambridge-accounts PROPERTIES AUTOMOC ON)
target_include_directories(streambridge-accounts PUBLIC src)
target_link_libraries(streambridge-accounts PUBLIC Qt6::Core Qt6::Network PRIVATE crypt32)
target_compile_definitions(streambridge-accounts PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(streambridge-updates STATIC src/updates/checker.cpp src/updates/launch.cpp)
set_target_properties(streambridge-updates PROPERTIES AUTOMOC ON)
target_include_directories(streambridge-updates PUBLIC src)
target_link_libraries(streambridge-updates PUBLIC Qt6::Core Qt6::Network PRIVATE shell32)
target_compile_definitions(streambridge-updates PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_definitions(streambridge-updates PRIVATE STREAMBRIDGE_VERSION="${PROJECT_VERSION}" STREAMBRIDGE_GITHUB_REPOSITORY="${STREAMBRIDGE_GITHUB_REPOSITORY}")
add_library(streambridge-proxy STATIC src/proxy/adapter.cpp)
set_target_properties(streambridge-proxy PROPERTIES AUTOMOC ON)
target_include_directories(streambridge-proxy PUBLIC src)
target_link_libraries(streambridge-proxy PUBLIC Qt6::Core Qt6::Network)
add_library(streambridge-browser-host STATIC src/browser/host.cpp src/browser/settings.cpp src/browser/probe.cpp)
set_target_properties(streambridge-browser-host PROPERTIES AUTOMOC ON)
target_link_libraries(streambridge-browser-host PUBLIC streambridge-proxy streambridge-accounts)
add_library(streambridge-runtime STATIC
  src/stream/encoder.cpp src/stream/output-manager.cpp)
set_target_properties(streambridge-runtime PROPERTIES AUTOMOC ON)
target_link_libraries(streambridge-runtime PUBLIC streambridge-core streambridge-accounts OBS::libobs Qt6::Core Qt6::Network
  PRIVATE crypt32)
target_compile_definitions(streambridge-runtime PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(streambridge-platforms STATIC src/platform/hub.cpp src/platform/irc.cpp)
set_target_properties(streambridge-platforms PROPERTIES AUTOMOC ON)
target_link_libraries(streambridge-platforms PUBLIC streambridge-accounts)
add_library(obs-streambridge MODULE src/plugin.cpp src/ui/dock.cpp src/ui/settings.cpp src/ui/account-panel.cpp
  src/ui/locale.cpp src/ui/platform-docks.cpp
  src/browser/source.cpp src/browser/worker.cpp)
set_target_properties(obs-streambridge PROPERTIES AUTOMOC ON PREFIX "")
target_link_libraries(obs-streambridge PRIVATE streambridge-runtime streambridge-browser-host streambridge-updates streambridge-platforms OBS::obs-frontend-api Qt6::Widgets)
target_compile_definitions(obs-streambridge PRIVATE STREAMBRIDGE_VERSION="${PROJECT_VERSION}")
install(TARGETS obs-streambridge RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-streambridge)
install(FILES "$<TARGET_FILE:Qt6::QSchannelBackendPlugin>"
  DESTINATION data/obs-plugins/obs-streambridge/qt-plugins/tls)
install(FILES "${CMAKE_SOURCE_DIR}/.deps/obs-deps-qt6-2025-08-23-x64/licenses/qt6/LICENSE.GPL2"
  DESTINATION data/obs-plugins/obs-streambridge/qt-plugins)
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md SECURITY.md CHANGELOG.md DESTINATION data/obs-plugins/obs-streambridge)
install(DIRECTORY docs/ DESTINATION data/obs-plugins/obs-streambridge/docs)
if(BUILD_TESTING)
  add_executable(security-tests tests/security-tests.cpp src/security/secret-store.cpp)
  target_link_libraries(security-tests PRIVATE Qt6::Core crypt32)
  target_include_directories(security-tests PRIVATE src)
  target_compile_definitions(security-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_test(NAME security COMMAND security-tests)
  add_executable(oauth-tests tests/oauth-tests.cpp)
  target_link_libraries(oauth-tests PRIVATE streambridge-accounts)
  add_test(NAME oauth COMMAND oauth-tests)
  add_executable(platform-tests tests/platform-tests.cpp)
  target_link_libraries(platform-tests PRIVATE streambridge-platforms)
  add_test(NAME platforms COMMAND platform-tests)
  add_executable(proxy-tests tests/proxy-tests.cpp)
  target_link_libraries(proxy-tests PRIVATE streambridge-proxy)
  add_test(NAME proxy COMMAND proxy-tests)
  add_executable(update-tests tests/update-tests.cpp)
  target_link_libraries(update-tests PRIVATE streambridge-updates)
  add_test(NAME updates COMMAND update-tests)
  add_executable(update-host-fixture tests/update-host-fixture.cpp tests/update-host-fixture.rc)
  target_link_libraries(update-host-fixture PRIVATE streambridge-updates)
  target_compile_definitions(update-host-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  # Exercise the production updater against a published release as an older client.
  # This test executable is never installed or packaged.
  add_executable(update-release-probe tests/update-release-probe.cpp src/updates/checker.cpp src/updates/launch.cpp)
  set_target_properties(update-release-probe PROPERTIES AUTOMOC ON)
  target_include_directories(update-release-probe PRIVATE src)
  target_link_libraries(update-release-probe PRIVATE Qt6::Core Qt6::Network shell32)
  target_compile_definitions(update-release-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
    STREAMBRIDGE_VERSION="0.1.1" STREAMBRIDGE_GITHUB_REPOSITORY="AugustusNobilis/StreamBridge")
  add_executable(browser-settings-tests tests/browser-settings-tests.cpp)
  target_link_libraries(browser-settings-tests PRIVATE streambridge-browser-host)
  add_test(NAME browser-settings COMMAND browser-settings-tests)
endif()
if(STREAMBRIDGE_INTEGRATION_TESTS)
  add_executable(ui-tests tests/ui-tests.cpp src/ui/dock.cpp src/ui/settings.cpp src/ui/account-panel.cpp src/ui/platform-docks.cpp)
  target_compile_definitions(ui-tests PRIVATE STREAMBRIDGE_VERSION="${PROJECT_VERSION}")
  target_link_libraries(ui-tests PRIVATE streambridge-runtime streambridge-platforms streambridge-updates Qt6::Widgets)
  add_test(NAME ui COMMAND ui-tests "${CMAKE_SOURCE_DIR}/data/locale" "${CMAKE_BINARY_DIR}/ui-review")
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows")
  add_library(streambridge-frontend-test MODULE tests/frontend-test-module.cpp)
  set_target_properties(streambridge-frontend-test PROPERTIES PREFIX "")
  target_link_libraries(streambridge-frontend-test PRIVATE streambridge-runtime OBS::obs-frontend-api Qt6::Widgets)
  add_executable(obs-integration tests/obs-integration.cpp)
  target_link_libraries(obs-integration PRIVATE streambridge-runtime)
  add_executable(rtmps-integration tests/rtmps-integration.cpp)
  target_link_libraries(rtmps-integration PRIVATE streambridge-runtime)
  add_executable(browser-probe tests/browser-probe.cpp)
  target_link_libraries(browser-probe PRIVATE streambridge-browser-host)
  add_executable(browser-media-tests tests/browser-media-tests.cpp src/browser/worker.cpp)
  set_target_properties(browser-media-tests PROPERTIES AUTOMOC ON)
  target_link_libraries(browser-media-tests PRIVATE streambridge-browser-host streambridge-runtime)
  add_executable(tls-widget-tests tests/tls-widget-tests.cpp)
  target_link_libraries(tls-widget-tests PRIVATE streambridge-browser-host)
endif()
