cmake_minimum_required(VERSION 3.28)
project(streambridge-browser LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "CEF minimal distribution configuration" FORCE)
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox is mandatory" FORCE)
set(USE_ATL OFF CACHE BOOL "No window UI" FORCE)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the pinned CEF binary distribution.")
endif()
file(TO_CMAKE_PATH "${CEF_ROOT}" CEF_ROOT)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_RELEASE}" "${CEF_LIB_RELEASE}")
add_library(streambridge-browser SHARED main.cpp writer.cpp)
SET_LIBRARY_TARGET_PROPERTIES(streambridge-browser)
target_include_directories(streambridge-browser PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../src")
target_link_libraries(streambridge-browser PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
set(CEF_TARGET_OUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIGURATION>")
COPY_SINGLE_FILE(streambridge-browser "${CEF_BINARY_DIR}/bootstrapc.exe" "${CEF_TARGET_OUT_DIR}/streambridge-browser.exe")
COPY_FILES(streambridge-browser "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
COPY_FILES(streambridge-browser "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
# Required read/execute access for the Chromium LPAC sandbox, limited to the helper build directory.
SET_LPAC_ACLS(streambridge-browser)
option(STREAMBRIDGE_TEST_FIXTURE "Build a separate synthetic widget helper (never package it)" OFF)
if(STREAMBRIDGE_TEST_FIXTURE)
  add_library(streambridge-browser-fixture SHARED main.cpp writer.cpp test-fixture.cpp)
  SET_LIBRARY_TARGET_PROPERTIES(streambridge-browser-fixture)
  target_compile_definitions(streambridge-browser-fixture PRIVATE STREAMBRIDGE_TEST_FIXTURE)
  target_include_directories(streambridge-browser-fixture PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../src")
  target_link_libraries(streambridge-browser-fixture PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  COPY_SINGLE_FILE(streambridge-browser-fixture "${CEF_BINARY_DIR}/bootstrapc.exe" "${CEF_TARGET_OUT_DIR}/streambridge-browser-fixture.exe")
endif()
